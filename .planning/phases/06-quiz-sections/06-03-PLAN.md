**PuzzlePanel.tsx — Operator adaptive zone for puzzle section:**

The puzzle section has these operator actions:
- View puzzle question text (preview in operator panel)
- Start/stop countdown timer using puzzle's configured duration from `data.parts.puzzles[puzzleIndex].duration`
- Mark "first team solved" → apply +15 pts to active team (using existing `addActiveScore` pattern from scoring presets)
- If second team also solves → instead apply +10 to first solver, +5 to second — operator manages this manually via number hotkeys (0 for +10, 5 for +5)

Local state: `puzzleIndex` (which puzzle in the episode, default 0) and `firstSolvedBy: 'right' | 'left' | null`.

```typescript
import { useState } from 'react'
import { useShowStore, useTimerStore } from '@/state'
import { useHotkeys } from 'react-hotkeys-hook'

export function PuzzlePanel() {
  const currentSection = useShowStore((s) => s.currentSection)
  const data = useShowStore((s) => s.data)
  const sectionState = useShowStore((s) => s.sectionState)
  const rightsTurn = useShowStore((s) => s.rightsTurn)
  const isActive = currentSection === 'puzzle'

  const [puzzleIndex, setPuzzleIndex] = useState(0)
  const [firstSolvedBy, setFirstSolvedBy] = useState<'right' | 'left' | null>(null)

  const puzzles = data?.parts.puzzles ?? []
  const currentPuzzle = puzzles[puzzleIndex]
  const duration = currentPuzzle?.duration ?? 90

  // T: toggle timer (start/stop). Wires to existing timerStore.
  useHotkeys('t', () => {
    const timerStore = useTimerStore.getState()
    if (!timerStore.countdownRunning) {
      // Set duration from puzzle data then start
      timerStore.setCountdown(duration)
      timerStore.setCountdownRunning(true)
    } else {
      timerStore.setCountdownRunning(false)
    }
  }, { enabled: isActive, enableOnFormTags: false }, [isActive, duration])

  // Shift+T: reset timer to puzzle duration
  useHotkeys('shift+t', () => {
    useTimerStore.getState().setCountdownRunning(false)
    useTimerStore.getState().setCountdown(duration)
  }, { enabled: isActive, enableOnFormTags: false }, [isActive, duration])

  // N: next puzzle
  useHotkeys('n, arrowright', () => {
    if (puzzleIndex + 1 < puzzles.length) {
      setPuzzleIndex((i) => i + 1)
      setFirstSolvedBy(null)
      useTimerStore.getState().setCountdownRunning(false)
    }
  }, { enabled: isActive, enableOnFormTags: false }, [isActive, puzzleIndex, puzzles.length])

  // B: previous puzzle
  useHotkeys('b, arrowleft', () => {
    if (puzzleIndex > 0) {
      setPuzzleIndex((i) => i - 1)
      setFirstSolvedBy(null)
    }
  }, { enabled: isActive, enableOnFormTags: false }, [isActive, puzzleIndex])

  // Enter: reveal answer on audience
  useHotkeys('enter', () => {
    const state = useShowStore.getState().sectionState
    useShowStore.getState().setSectionState({ answerRevealed: !state.answerRevealed })
  }, { enabled: isActive, enableOnFormTags: false }, [isActive])

  // "Mark first solve" button handler (also callable via hotkey):
  const markFirstSolve = () => {
    const team = rightsTurn ? 'right' : 'left'
    setFirstSolvedBy(team)
    // Apply +15 pts to active team
    if (rightsTurn) useShowStore.getState().addRightScore(15)
    else useShowStore.getState().addLeftScore(15)
    useTimerStore.getState().setCountdownRunning(false)
  }

  // "Both solved" — reverts single solve and applies split (operator pressed 0 then 5 manually via existing hotkeys)
  // Operator uses existing scoring hotkeys (0 = +10, 5 = +5) for split scoring
  // Just show instructions in the panel

  if (!isActive) return null

  return (
    <div className="space-y-2 pt-1">
      <div className="flex items-center justify-between">
        <span className="text-xs font-medium text-muted-foreground">اللغز</span>
        <span className="text-xs font-bold tabular-nums western-numerals bg-muted px-2 py-0.5 rounded">
          {puzzleIndex + 1} / {puzzles.length}
        </span>
      </div>

      <div className="rounded border border-border bg-card p-2 text-sm text-right leading-relaxed min-h-[3rem]">
        {currentPuzzle?.text ?? <span className="text-muted-foreground text-xs">لا يوجد لغز</span>}
      </div>

      {sectionState.answerRevealed && currentPuzzle?.answer && (
        <div className="rounded border border-green-500/30 bg-green-500/10 p-2 text-xs text-right text-green-400 whitespace-pre-line leading-relaxed">
          {currentPuzzle.answer}
        </div>
      )}

      <div className="rounded border border-border bg-muted/30 p-2 text-[10px] text-muted-foreground space-y-1">
        <p>المدة: <span className="font-bold western-numerals text-foreground">{duration}</span> ثانية</p>
        {firstSolvedBy && (
          <p className="text-amber-400">
            حلّ أولاً: {firstSolvedBy === 'right' ? 'الأيمن' : 'الأيسر'} (+15 ✓)
          </p>
        )}
        {firstSolvedBy && (
          <p className="text-muted-foreground">إذا حلّ الثاني: استخدم <kbd className="bg-muted rounded px-1">0</kbd>+10 و<kbd className="bg-muted rounded px-1">5</kbd>+5</p>
        )}
      </div>

      <div className="grid grid-cols-2 gap-1">
        <button
          onClick={() => {
            const timerStore = useTimerStore.getState()
            if (!timerStore.countdownRunning) {
              timerStore.setCountdown(duration)
              timerStore.setCountdownRunning(true)
            } else {
              timerStore.setCountdownRunning(false)
            }
          }}
          className="rounded border border-border bg-card px-2 py-1.5 text-[11px] hover:bg-accent transition-colors"
        >
          بدء/إيقاف <kbd className="text-[9px] bg-muted rounded px-1">T</kbd>
        </button>
        <button
          onClick={markFirstSolve}
          disabled={!!firstSolvedBy}
          className="rounded border border-amber-500/40 bg-amber-500/10 px-2 py-1.5 text-[11px] hover:bg-amber-500/20 transition-colors disabled:opacity-40"
        >
          حلّ أولاً (+15)
        </button>
      </div>
    </div>
  )
}
```

**RapidQuestionsPanel.tsx — Operator adaptive zone for rapid-questions section:**

Data structure: `data.parts.quickQuestions` is an array of `{title: string, questions: {text, answer, ...}[]}`.
Each QuickQuestionSet = one round for one team. Operator navigates through question sets.

Section mechanics:
- `rapidActiveTeam` in sectionState tracks which team is currently answering
- S key: switch team — pauses timer, resets countdown to 60s, waits for operator to start
- T key: start/stop timer (reuses existing timerStore countdown)
- N/ArrowRight: next question within current set
- B/ArrowLeft: previous question within current set
- Enter: reveal/hide answer (same pattern)

The audience display always shows the current question text (same questions for both teams — the idea is both teams answer the same 20 questions). `questionIndex` in sectionState tracks the current question within the active question set.

```typescript
import { useShowStore, useTimerStore } from '@/state'
import { useHotkeys } from 'react-hotkeys-hook'

const RAPID_TIMER_SECONDS = 60

export function RapidQuestionsPanel() {
  const currentSection = useShowStore((s) => s.currentSection)
  const data = useShowStore((s) => s.data)
  const sectionState = useShowStore((s) => s.sectionState)
  const rightsTurn = useShowStore((s) => s.rightsTurn)
  const isActive = currentSection === 'rapid-questions'

  const quickSets = data?.parts.quickQuestions ?? []
  // Use questionIndex as the set index (which question SET we're on)
  // Sub-question index tracked separately — add rapidSubIndex to sectionState, or use local state
  // Per research: Rapid Questions is "same set of 20 questions" — one set, operator navigates sub-questions
  // Looking at data: quickQuestions[i].questions[] has the individual questions
  // questionIndex = which set (0-based), rapidSubIndex = which question within set
  // Since rapidSubIndex isn't in sectionState yet, add it. But to avoid modifying showStore again,
  // use local state for sub-question index (operator-only concern, audience needs set + sub index for question text).
  // DECISION: Track set index (sectionState.questionIndex) + sub-question via sectionState.
  // Add rapidSubIndex to sectionState in this plan's showStore modification.
  // SIMPLER: Use questionIndex for the sub-question index within the first/only set.
  // The data shows quickQuestions[0].title = "ما؟" with 20 sub-questions.
  // There is typically one set per episode section run. Use questionIndex for sub-question.
  // rapidActiveTeam in sectionState tracks who's answering.

  const activeTeam = sectionState.rapidActiveTeam
  const currentSet = quickSets[0] // Use first set (adjust if multiple sets exist)
  const subQuestions = currentSet?.questions ?? []
  const currentSubQ = subQuestions[sectionState.questionIndex]
  const totalSubQ = subQuestions.length

  // Initialize active team to right on first render
  // (handled by operator pressing S the first time to pick a team, or via button)

  // S: switch active team — pause timer, reset to 60s
  useHotkeys('s', () => {
    useTimerStore.getState().setCountdownRunning(false)
    useTimerStore.getState().setCountdown(RAPID_TIMER_SECONDS)
    const current = useShowStore.getState().sectionState.rapidActiveTeam
    const next = current === 'right' ? 'left' : 'right'
    useShowStore.getState().setSectionState({
      rapidActiveTeam: next,
      questionIndex: 0,    // reset question index for new team
      answerRevealed: false,
    })
    // Also update rightsTurn in showStore to reflect active team (visual indicator)
    useShowStore.getState().setRightsTurn(next === 'right')
    useShowStore.getState().setTurned(true)
  }, { enabled: isActive, enableOnFormTags: false }, [isActive])

  // T: start/stop timer
  useHotkeys('t', () => {
    const timerStore = useTimerStore.getState()
    if (!timerStore.countdownRunning) {
      if (timerStore.countdownRemaining === 0) {
        timerStore.setCountdown(RAPID_TIMER_SECONDS)
      }
      timerStore.setCountdownRunning(true)
    } else {
      timerStore.setCountdownRunning(false)
    }
  }, { enabled: isActive, enableOnFormTags: false }, [isActive])

  // N / ArrowRight: next sub-question
  useHotkeys('n, arrowright', () => {
    const state = useShowStore.getState().sectionState
    const newIndex = Math.min(state.questionIndex + 1, totalSubQ - 1)
    useShowStore.getState().setSectionState({ questionIndex: newIndex, answerRevealed: false })
  }, { enabled: isActive, enableOnFormTags: false }, [isActive, totalSubQ])

  // B / ArrowLeft: previous sub-question
  useHotkeys('b, arrowleft', () => {
    const state = useShowStore.getState().sectionState
    useShowStore.getState().setSectionState({ questionIndex: Math.max(0, state.questionIndex - 1), answerRevealed: false })
  }, { enabled: isActive, enableOnFormTags: false }, [isActive])

  // Enter: reveal/hide answer
  useHotkeys('enter', () => {
    const state = useShowStore.getState().sectionState
    useShowStore.getState().setSectionState({ answerRevealed: !state.answerRevealed })
  }, { enabled: isActive, enableOnFormTags: false }, [isActive])

  if (!isActive) return null

  return (
    <div className="space-y-2 pt-1">
      {/* Active team indicator */}
      <div className="flex items-center justify-between">
        <span className="text-xs font-medium text-muted-foreground">الرشق السريع</span>
        <span className="text-xs font-bold tabular-nums western-numerals bg-muted px-2 py-0.5 rounded">
          {sectionState.questionIndex + 1} / {totalSubQ}
        </span>
      </div>

      {/* Set title */}
      {currentSet?.title && (
        <div className="rounded border border-border bg-muted/30 px-2 py-1 text-xs text-center font-medium">
          {currentSet.title}
        </div>
      )}

      {/* Active team banner */}
      <div className={
        'rounded border px-2 py-1.5 text-[11px] text-center font-medium ' +
        (activeTeam === 'right'
          ? 'border-primary/40 bg-primary/10 text-primary'
          : activeTeam === 'left'
            ? 'border-amber-500/40 bg-amber-500/10 text-amber-500'
            : 'border-border bg-card text-muted-foreground')
      }>
        {activeTeam
          ? `يجيب: ${activeTeam === 'right' ? 'الفريق الأيمن' : 'الفريق الأيسر'}`
          : 'اضغط S لاختيار الفريق'}
      </div>

      {/* Current question */}
      <div className="rounded border border-border bg-card p-2 text-sm text-right leading-relaxed min-h-[3rem]">
        {currentSubQ?.text ?? <span className="text-muted-foreground text-xs">لا يوجد سؤال</span>}
      </div>

      {/* Controls */}
      <div className="grid grid-cols-2 gap-1">
        <button
          onClick={() => {
            useTimerStore.getState().setCountdownRunning(false)
            useTimerStore.getState().setCountdown(RAPID_TIMER_SECONDS)
            const current = useShowStore.getState().sectionState.rapidActiveTeam
            const next = current === 'right' ? 'left' : 'right'
            useShowStore.getState().setSectionState({ rapidActiveTeam: next, questionIndex: 0, answerRevealed: false })
            useShowStore.getState().setRightsTurn(next === 'right')
            useShowStore.getState().setTurned(true)
          }}
          className="rounded border border-border bg-card px-2 py-1.5 text-[11px] hover:bg-accent transition-colors"
        >
          تبديل الفريق <kbd className="text-[9px] bg-muted rounded px-1">S</kbd>
        </button>
        <button
          onClick={() => {
            const timerStore = useTimerStore.getState()
            if (!timerStore.countdownRunning) {
              if (timerStore.countdownRemaining === 0) timerStore.setCountdown(RAPID_TIMER_SECONDS)
              timerStore.setCountdownRunning(true)
            } else {
              timerStore.setCountdownRunning(false)
            }
          }}
          className="rounded border border-border bg-card px-2 py-1.5 text-[11px] hover:bg-accent transition-colors"
        >
          بدء/إيقاف <kbd className="text-[9px] bg-muted rounded px-1">T</kbd>
        </button>
      </div>
    </div>
  )
}
```
